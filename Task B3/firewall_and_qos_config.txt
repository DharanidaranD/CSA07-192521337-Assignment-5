tc qdisc del dev eth0 root                               # reset after each test

# ---- B3.3 : corrected firewall rules (iptables, order matters) ----
iptables -F INPUT
iptables -A INPUT -m conntrack --ctstate ESTABLISHED,RELATED -j ACCEPT   # stateful improvement
iptables -A INPUT -i lo -j ACCEPT
iptables -A INPUT -p tcp -s 10.1.0.0/16 --dport 22 -j ACCEPT
iptables -A INPUT -p tcp --dport 22 -j DROP
iptables -A INPUT -p tcp --dport 443 -j ACCEPT
iptables -A INPUT -p udp --dport 53 -j ACCEPT
iptables -P INPUT DROP
# Tests: nmap -p 22,443 <srv> from outside address (22 filtered, 443 open); ssh from 10.1.x.x works.
